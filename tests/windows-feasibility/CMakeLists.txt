cmake_minimum_required(VERSION 3.24)
project(EveriaFilesystemFeasibility LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This isolated experiment requires native Windows x64/MSVC")
endif()
add_executable(ntfs-feasibility feasibility.cpp)
target_compile_features(ntfs-feasibility PRIVATE cxx_std_20)
target_compile_definitions(ntfs-feasibility PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ntfs-feasibility PRIVATE /W4 /WX /EHsc)
target_link_libraries(ntfs-feasibility PRIVATE bcrypt)
add_executable(rename-review rename-review.cpp)
target_compile_features(rename-review PRIVATE cxx_std_20)
target_compile_definitions(rename-review PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(rename-review PRIVATE /W4 /WX /EHsc)
target_link_libraries(rename-review PRIVATE bcrypt)
