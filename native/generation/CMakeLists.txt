cmake_minimum_required(VERSION 3.24)
project(EveriaGeneration LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Native Windows x64/MSVC is required; no fallback")
endif()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
foreach(var EVERIA_PAYLOAD_DIR EVERIA_SOURCE_SHA EVERIA_BUILD_ID EVERIA_ARCHIVE_SHA)
  if(NOT DEFINED ${var})
    message(FATAL_ERROR "Missing immutable payload provenance: ${var}")
  endif()
endforeach()
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/generation/inventory.py"
  --payload "${EVERIA_PAYLOAD_DIR}" --source "${EVERIA_SOURCE_SHA}" --build "${EVERIA_BUILD_ID}"
  --archive "${EVERIA_ARCHIVE_SHA}" --output "${GENERATED}" RESULT_VARIABLE inventory_result COMMAND_ERROR_IS_FATAL ANY)
if(NOT inventory_result EQUAL 0)
  message(FATAL_ERROR "Inventory refused")
endif()
function(configure_engine target)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc)
  target_include_directories(${target} PRIVATE "${GENERATED}" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(${target} PRIVATE bcrypt shell32 ole32)
endfunction()
add_library(everia-generation STATIC generation.cpp inventory_binding.cpp)
configure_engine(everia-generation)
option(EVERIA_GENERATION_TESTS "Build isolated production-engine tests" OFF)
if(EVERIA_GENERATION_TESTS)
  add_executable(generation-tests generation.cpp inventory_binding.cpp ../../tests/generation/native_tests.cpp)
  configure_engine(generation-tests)
  target_compile_definitions(generation-tests PRIVATE EVERIA_GENERATION_TESTING)
endif()
